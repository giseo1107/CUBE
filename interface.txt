*커스텀큐브(customCube.jsp)

큐브 불러오기 (cubeInfo.jsp)

DB

-- 1. 도면(프로젝트) 정보를 담을 테이블
CREATE TABLE projects (
    id INT AUTO_INCREMENT PRIMARY KEY,
    title VARCHAR(100) NOT NULL,
    canvas_width INT DEFAULT 600,
    canvas_height INT DEFAULT 600,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 2. 도면 안에 배치된 개별 가구 정보를 담을 테이블
CREATE TABLE furniture_items (
    id INT AUTO_INCREMENT PRIMARY KEY,
    project_id INT,               -- 어느 도면에 속하는지 (Foreign Key)
    obj_name VARCHAR(50),         -- 가구 이름 (예: Bed_123)
    pos_x INT,                    -- X 좌표 (left)
    pos_y INT,                    -- Y 좌표 (top)
    width INT,                    -- 가로 길이 (실제 cm)
    depth INT,                    -- 세로 길이 (실제 cm)
    height INT,                   -- 높이 (두께)
    elevation INT,                -- 공중 부양 높이
    rotation INT,                 -- 회전 각도
    color VARCHAR(20),            -- 색상 코드
    FOREIGN KEY (project_id) REFERENCES projects(id) ON DELETE CASCADE
);
